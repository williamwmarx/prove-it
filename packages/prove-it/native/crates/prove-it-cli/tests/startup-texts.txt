IsSimpleCycle(S, r) ==
  LET F[i \in 1..Cardinality(S)] ==
        IF i = 1 THEN << CHOOSE s \in S : TRUE >>

  IN Range(F[Cardinality(S)]) = S
RingOfNodes == CHOOSE r \in [Node -> Node] : IsSimpleCycle(Node, r)
